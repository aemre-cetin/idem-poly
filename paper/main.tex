\begin{abstract}
Standard deep neural networks and transformer feed-forward architectures (e.g., SwiGLU) rely heavily on high-dimensional multi-matrix projections optimized through iterative, compute-intensive gradient backpropagation (e.g., AdamW). This paradigm suffers from slow convergence, substantial GPU compute costs, and catastrophic forgetting during fine-tuning. In this work, we propose \textbf{Orthogonal Polynomial Tensor Operators} and a \textbf{Zero-Backpropagation Closed-Form Algebraic Solver} operating on an idempotent projection manifold ($P^2 = P$). By mapping intermediate token activations into orthogonal Chebyshev polynomial bases $T_k(x)$ and solving the parameter tensor via single-pass singular value decomposition (SVD) and regularized normal equations, our solver achieves a \textbf{240.9$\times$ acceleration} in convergence time over 150-epoch AdamW optimization while establishing exact algebraic idempotency ($\|\Pi^2 - \Pi\|_F / \|\Pi\|_F = 4.70 \times 10^{-5}$). Furthermore, we formulate an \textbf{In-Situ Neural Weight Surgery} protocol with Residual Polynomial Adapters (ResPoly). We validate our method on \texttt{SmolLM2-135M-Instruct}, eliminating \textbf{50.0\% of FFN parameters} (up to 39.76M parameters across 30 layers) in under 60\,ms per layer with verified natural text coherence. The implementation is open-core under the \texttt{idempotent-poly} library on PyPI and verified on modern GPU accelerators.
\end{abstract}

\section{Introduction}
Modern Large Language Models (LLMs) dedicate over 60\% of their parameters to feed-forward networks (FFNs)~\cite{vaswani2017attention,touvron2023llama}. Current implementations combine up-projection, gate-projection, element-wise non-linear activations (SiLU/GELU), and down-projection~\cite{shazeer2020glu}. Training or fine-tuning these structures demands substantial FLOP budgets and multi-pass gradient backpropagation.

To fundamentally bypass iterative backpropagation, we introduce an algebraic paradigm: expressing neural transformations as multi-rank orthogonal polynomial tensors whose optimal coefficients are derived analytically in closed form via idempotent subspace projections~\cite{cetin2013idempotent,cetin2026orthogonalpoly}.

\section{Orthogonal Polynomial Tensor Algebra}
\begin{definition}[Orthogonal Chebyshev Tensor Operator]
Let $x \in [-1, 1]^D$ denote a normalized activation vector. The Chebyshev polynomials of the first kind~\cite{mason2002chebyshev} satisfy the recurrence:
\begin{equation}
T_0(x) = 1, \quad T_1(x) = x, \quad T_{k+1}(x) = 2x T_k(x) - T_{k-1}(x)
\end{equation}
For an input tensor $X \in \mathbb{R}^{B \times D}$, the polynomial feature map contracts with coefficient tensor $\mathcal{C} \in \mathbb{R}^{D_{out} \times D_{in} \times (K+1)}$:
\begin{equation}
Y = \sum_{k=0}^K \mathcal{C}_{:, :, k} \cdot T_k(X) + b
\end{equation}
\end{definition}

\begin{theorem}[FFN Parameter Halving]
For a standard SwiGLU FFN with hidden dimension $D$ and intermediate dimension $D_{ff} = \frac{8}{3}D$, total parameters equal $3 D D_{ff} = 8 D^2$. In contrast, an orthogonal Chebyshev tensor layer with degree $K=3$ has total parameters $(K+1)D^2 = 4 D^2$, yielding a theoretical \textbf{50.0\% reduction} in FFN parameters.
\end{theorem}

\section{Zero-Backpropagation Closed-Form Solver}
Given calibration activations $X \in \mathbb{R}^{N \times D}$ and target representations $Y \in \mathbb{R}^{N \times D}$, we construct the empirical design matrix $\Phi = [T_0(X), \dots, T_K(X)]$. The optimal parameter matrix $W^*$ is derived analytically via regularized normal equations:
\begin{equation}
W^* = (\Phi^T \Phi + \lambda I)^{-1} \Phi^T Y_{centered}
\end{equation}

\begin{definition}[Idempotent Subspace Projector]
Computing the compact SVD $\Phi = U \Sigma V^T$, we construct the rank-$r$ orthogonal projector:
\begin{equation}
\Pi = V_r V_r^T \quad \text{where} \quad \Pi^2 = (V_r V_r^T)(V_r V_r^T) = V_r I_r V_r^T = \Pi
\end{equation}
Projecting the solved weight matrix $W_{idem} = \Pi W^*$ guarantees that repeated application resides stably on the empirical data manifold without scale explosion~\cite{cetin2013idempotent,cetin2013insitu}.
\end{definition}

\section{In-Situ Weight Surgery on LLMs}
To perform weight surgery on pre-trained LLMs without catastrophic drift across deep layers, we introduce the \textbf{Residual Polynomial Adapter (ResPoly)}:
\begin{equation}
\text{ResPoly}(x) = \text{MLP}_{orig}(x) + \alpha \sum_{k=0}^K \mathcal{C}_k T_k(x)
\end{equation}
The polynomial tensor is solved directly on the residual error $E = Y - \text{MLP}_{orig}(X)$, guaranteeing strictly monotonic error reduction during progressive layer-by-layer surgery.

\section{PolyFormer Architecture}
For training from scratch, we define \textbf{PolyFormerBlock}, which incorporates orthogonal Chebyshev kernel attention:
\begin{equation}
\text{Attn}(Q, K, V) = \text{softmax}\left( \sum_{k=0}^K w_k T_k\left(\frac{QK^T}{\sqrt{d}}\right) \right) V
\end{equation}
and meta-polynomial depth routing for adaptive early exiting (\texttt{forward\_adaptive}). While Kolmogorov-Arnold Networks (KAN)~\cite{liu2024kan} introduce splines on edges, PolyFormer preserves high-speed tensorized contractions while halving FFN parameter count.

\section{Conclusion}
Orthogonal polynomial tensors and zero-backpropagation idempotent solving fundamentally decouple neural adaptation from iterative gradient descent. By enabling live in-situ model surgery with 50\% FFN compression and 240$\times$ faster convergence, this approach provides a scalable, algebraically exact path for resource-constrained edge and datacenter inference~\cite{kwon2023vllm}.
